# test_load.py
import asyncio
import aiohttp
import time
import statistics

async def make_request(session, url, headers):
    """发送单个请求"""
    try:
        async with session.get(url, headers=headers) as response:
            return response.status, response.elapsed.total_seconds()
    except Exception as e:
        return None, None

async def load_test(concurrent_users, duration_seconds):
    """负载测试"""
    url = "http://localhost:8000/api/records/"
    headers = {"Authorization": "Bearer test_token"}
    
    results = []
    start_time = time.time()
    
    async with aiohttp.ClientSession() as session:
        while time.time() - start_time < duration_seconds:
            tasks = []
            for _ in range(concurrent_users):
                task = asyncio.create_task(make_request(session, url, headers))
                tasks.append(task)
            
            batch_results = await asyncio.gather(*tasks)
            results.extend(batch_results)
            
            # 短暂休息，避免过度请求
            await asyncio.sleep(0.1)
    
    # 分析结果
    successful_requests = [r for r in results if r[0] == 200]
    failed_requests = [r for r in results if r[0] != 200]
    response_times = [r[1] for r in successful_requests if r[1] is not None]
    
    print(f"测试结果:")
    print(f"  并发用户数: {concurrent_users}")
    print(f"  测试时长: {duration_seconds}秒")
    print(f"  总请求数: {len(results)}")
    print(f"  成功请求数: {len(successful_requests)}")
    print(f"  失败请求数: {len(failed_requests)}")
    print(f"  成功率: {len(successful_requests)/len(results)*100:.2f}%")
    
    if response_times:
        print(f"  平均响应时间: {statistics.mean(response_times)*1000:.2f}毫秒")
        print(f"  中位数响应时间: {statistics.median(response_times)*1000:.2f}毫秒")
        print(f"  95%响应时间: {sorted(response_times)[int(len(response_times)*0.95)]*1000:.2f}毫秒")
        print(f"  99%响应时间: {sorted(response_times)[int(len(response_times)*0.99)]*1000:.2f}毫秒")
        print(f"  吞吐量: {len(successful_requests)/duration_seconds:.2f}请求/秒")

# 运行测试
if __name__ == "__main__":
    asyncio.run(load_test(concurrent_users=50, duration_seconds=60))

#### 安全扫描脚本
```python
# security_audit.py
import requests
import json
import re

class SecurityAuditor:
    def __init__(self, base_url):
        self.base_url = base_url
        self.findings = []
    
    def test_sql_injection(self):
        """测试SQL注入"""
        # 注意：ChromaDB不是SQL数据库，但仍需测试参数化查询
        test_cases = [
            {"username": "admin' OR '1'='1", "password": "test"},
            {"username": "admin'; DROP TABLE users; --", "password": "test"},
        ]
        
        for test_case in test_cases:
            response = requests.post(f"{self.base_url}/api/auth/login", json=test_case)
            if response.status_code == 200:
                self.findings.append({
                    "type": "SQL_INJECTION",
                    "severity": "HIGH",
                    "description": f"潜在的SQL注入漏洞: {test_case}",
                    "url": "/api/auth/login"
                })
    
    def test_xss(self):
        """测试XSS攻击"""
        test_cases = [
            {"username": "<script>alert('xss')</script>", "password": "test"},
            {"username": "test<img src=x onerror=alert('xss')>", "password": "test"},
        ]
        
        for test_case in test_cases:
            response = requests.post(f"{self.base_url}/api/auth/register", json=test_case)
            if response.status_code == 200:
                # 检查响应中是否包含未转义的HTML
                if "<script>" in response.text or "onerror=" in response.text:
                    self.findings.append({
                        "type": "XSS",
                        "severity": "MEDIUM",
                        "description": f"潜在的XSS漏洞: {test_case}",
                        "url": "/api/auth/register"
                    })
    
    def test_csrf(self):
        """测试CSRF攻击"""
        # 检查是否缺少CSRF保护
        response = requests.post(f"{self.base_url}/api/auth/login", json={
            "username": "admin",
            "password": "admin123"
        })
        
        # 检查是否缺少CSRF令牌
        if "csrf" not in response.text.lower() and "token" not in response.text.lower():
            self.findings.append({
                "type": "CSRF",
                "severity": "MEDIUM",
                "description": "缺少CSRF保护",
                "url": "/api/auth/login"
            })
    
    def test_authentication(self):
        """测试认证安全"""
        # 测试未认证访问受保护资源
        protected_urls = [
            "/api/records/",
            "/api/admin/dashboard",
            "/api/detection/models"
        ]
        
        for url in protected_urls:
            response = requests.get(f"{self.base_url}{url}")
            if response.status_code != 401:
                self.findings.append({
                    "type": "AUTHENTICATION",
                    "severity": "HIGH",
                    "description": f"未认证可访问受保护资源: {url}",
                    "url": url
                })
    
    def test_sensitive_data(self):
        """测试敏感数据泄露"""
        # 检查是否泄露敏感信息
        response = requests.get(f"{self.base_url}/health")
        
        # 检查是否包含敏感信息
        sensitive_patterns = [
            r"password",
            r"secret",
            r"key",
            r"token",
            r"database"
        ]
        
        for pattern in sensitive_patterns:
            if re.search(pattern, response.text, re.IGNORECASE):
                self.findings.append({
                    "type": "SENSITIVE_DATA",
                    "severity": "MEDIUM",
                    "description": f"可能泄露敏感信息: {pattern}",
                    "url": "/health"
                })
    
    def run_audit(self):
        """运行安全审计"""
        print("开始安全审计...")
        
        self.test_sql_injection()
        self.test_xss()
        self.test_csrf()
        self.test_authentication()
        self.test_sensitive_data()
        
        print(f"\n审计完成，发现 {len(self.findings)} 个问题:")
        for finding in self.findings:
            print(f"  [{finding['severity']}] {finding['type']}: {finding['description']}")
        
        return self.findings

# 运行审计
if __name__ == "__main__":
    auditor = SecurityAuditor("http://localhost:8000")
    findings = auditor.run_audit()
```
